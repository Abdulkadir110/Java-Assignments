// for
for (int counter = 1; counter <= 10; counter++)	{
			System.out.println(counter);
		}


// While
int counter = 1;
		while(counter <= 10) {
			System.out.println(counter);
		counter++;
		}
	}

//Do while
int counter = 1;
do {
	System.out.println(counter);
	counter++;
    }
while(counter <= 10);
